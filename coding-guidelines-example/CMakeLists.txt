cmake_minimum_required(VERSION 3.24)
project(orbex LANGUAGES CXX)

# A standalone example. It builds on its own -- it is not part of the orbsim
# build -- so that it can be copied, compiled and poked at without dragging a
# Vulkan SDK along.

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

# ---------------------------------------------------------------------------
# [S1] Warnings, and warnings as errors.
#
# "The compiler is the cheapest code reviewer you will ever have." Carried as an
# INTERFACE target so every target below gets exactly the same set and nobody
# can quietly opt out of one.
# ---------------------------------------------------------------------------
add_library(orbex_warnings INTERFACE)
target_compile_options(orbex_warnings INTERFACE
        $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:
            -Wall -Wextra -Wpedantic
            -Wshadow
            -Wnon-virtual-dtor
            -Wold-style-cast
            -Wcast-align
            -Wunused
            -Woverloaded-virtual
            -Wconversion
            -Wsign-conversion
            -Wnull-dereference
            -Wdouble-promotion
            -Wformat=2
            -Wimplicit-fallthrough
            -Werror>
        $<$<CXX_COMPILER_ID:MSVC>:/W4 /WX /permissive->
)

# ---------------------------------------------------------------------------
# [S11] Floating-point settings.
#
# -ffp-contract=off: FMA contraction reorders arithmetic, and reordered
# arithmetic breaks the bit-identical reproduction that the determinism test
# asserts. -ffast-math is never set anywhere, at any optimization level.
# ---------------------------------------------------------------------------
add_library(orbex_fp INTERFACE)
target_compile_options(orbex_fp INTERFACE
        $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-ffp-contract=off>
        $<$<CXX_COMPILER_ID:MSVC>:/fp:precise>
)

# ---------------------------------------------------------------------------
# [S1] AddressSanitizer, off by default and one flag away.
#
# UBSan is deliberately not wired in here: its Windows support is partial, and
# a build option that silently does less than its name promises is worse than
# no option at all. On a Linux job, add -fsanitize=address,undefined.
# ---------------------------------------------------------------------------
option(ORBEX_SANITIZE_ADDRESS "Build with AddressSanitizer" OFF)
add_library(orbex_sanitizers INTERFACE)
if(ORBEX_SANITIZE_ADDRESS)
    target_compile_options(orbex_sanitizers INTERFACE
            $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-fsanitize=address -fno-omit-frame-pointer>)
    target_link_options(orbex_sanitizers INTERFACE
            $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-fsanitize=address>)
endif()

add_library(orbex_settings INTERFACE)
target_link_libraries(orbex_settings INTERFACE orbex_warnings orbex_fp orbex_sanitizers)

# ---------------------------------------------------------------------------
# [S12] The simulation core. Knows nothing of rendering.
#
# The layering is expressed as a dependency, not as a convention somebody has
# to remember: orbex_core cannot reference orbex_render, because it does not
# link it and CMake would fail if it tried.
# ---------------------------------------------------------------------------
add_library(orbex_core STATIC
        src/orbit/Kepler.cpp
        src/orbit/OrbitPath.cpp
)
target_include_directories(orbex_core PUBLIC src)
target_link_libraries(orbex_core PUBLIC orbex_fp PRIVATE orbex_warnings orbex_sanitizers)

# [S12] The renderer-side layer. Depends on core; core never depends on it.
add_library(orbex_render STATIC
        src/render/PathUpload.cpp
)
target_link_libraries(orbex_render PUBLIC orbex_core PRIVATE orbex_warnings orbex_sanitizers)

# ---------------------------------------------------------------------------
# [S14] SF.11, enforced mechanically.
#
# "Every header must compile on its own." That is a claim, and claims decay. A
# generated translation unit per header, containing nothing but that one
# #include, turns it into a build failure the moment it stops being true.
# ---------------------------------------------------------------------------
set(ORBEX_HEADERS
        core/Contract.hpp
        core/Vec3.hpp
        core/Units.hpp
        orbit/Kepler.hpp
        orbit/OrbitPath.hpp
        render/PathUpload.hpp
)

set(orbex_selfcheck_sources "")
foreach(header IN LISTS ORBEX_HEADERS)
    string(REPLACE "/" "_" stem "${header}")
    string(REPLACE "." "_" stem "${stem}")
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/selfcheck/selfcheck_${stem}.cpp")
    file(WRITE "${generated}"
            "// Generated by CMake. Proves ${header} is self-contained (SF.11).\n"
            "#include \"${header}\"\n")
    list(APPEND orbex_selfcheck_sources "${generated}")
endforeach()

add_library(orbex_header_selfcheck OBJECT ${orbex_selfcheck_sources})
target_link_libraries(orbex_header_selfcheck PRIVATE orbex_core orbex_warnings)

# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------
include(CTest)

add_library(orbex_test_support INTERFACE)
target_include_directories(orbex_test_support INTERFACE .)
target_link_libraries(orbex_test_support INTERFACE orbex_render orbex_warnings orbex_sanitizers)

foreach(test_name IN ITEMS test_units test_kepler test_orbit_path)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE orbex_test_support)
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

# The self-containment objects are not a test, they are a build step: if a
# header stops compiling alone, nothing links.
add_dependencies(test_units orbex_header_selfcheck)

# ---------------------------------------------------------------------------
# [S10] The benchmark.
#
# Built alongside the tests but deliberately NOT registered with CTest: a timing
# assertion on shared hardware fails for reasons unrelated to the code, and a
# test that cries wolf gets disabled.
# ---------------------------------------------------------------------------
add_executable(bench_orbit_path tests/bench_orbit_path.cpp)
target_link_libraries(bench_orbit_path PRIVATE orbex_core orbex_warnings)
